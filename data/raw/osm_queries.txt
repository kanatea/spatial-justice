HOSPITALS + EMERGENCY SERVICES

[out:json][timeout:180];
{{geocodeArea:Czechia}}->.searchArea;

(
  // 1. Find anything explicitly tagged as an emergency facility
  nwr["amenity"="hospital"](area.searchArea);
  nwr["emergency"="yes"](area.searchArea);
  nwr["emergency"="doctor"](area.searchArea);
  
  // 2. Find hospitals or clinics that have "urgent" or "emergency" in the name
  // "urgentní" = urgent/emergency in Czech
  nwr["amenity"~"hospital|clinic"]["name"~"urgentní|pohotovost",i](area.searchArea);
);

out body;
>;
out skel qt;
